cmake_minimum_required(VERSION 3.24)
project(eventAnalyzer VERSION 1.0.0 LANGUAGES C CXX)
include(CTest)
option(EA_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer (GCC/Clang)" OFF)
option(EA_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)

find_package(Threads REQUIRED)
add_library(ea_sqlite STATIC vendor/sqlite3/sqlite3.c)
target_include_directories(ea_sqlite PUBLIC vendor/sqlite3)
target_compile_definitions(ea_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_DQS=0)
target_link_libraries(ea_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(ea_xml STATIC vendor/pugixml/pugixml.cpp)
target_include_directories(ea_xml PUBLIC vendor/pugixml)

add_library(event_analyzer_core
    src/model.cpp src/xml_importer.cpp src/process_index.cpp
    src/detection_engine.cpp src/event_store.cpp)
target_include_directories(event_analyzer_core PUBLIC include)
target_compile_features(event_analyzer_core PUBLIC cxx_std_20)
target_link_libraries(event_analyzer_core PRIVATE ea_xml ea_sqlite)
add_executable(eventAnalyzer src/main.cpp)
target_link_libraries(eventAnalyzer PRIVATE event_analyzer_core)
if(MINGW)
    target_link_options(eventAnalyzer PRIVATE -municode)
endif()

set(ea_targets event_analyzer_core eventAnalyzer)
if(BUILD_TESTING)
    add_executable(eventAnalyzer_tests tests/test_core.cpp)
    target_link_libraries(eventAnalyzer_tests PRIVATE event_analyzer_core)
    list(APPEND ea_targets eventAnalyzer_tests)
    add_test(NAME core COMMAND eventAnalyzer_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:eventAnalyzer>")
    endif()
endif()
foreach(target IN LISTS ea_targets)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        if(EA_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(EA_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endforeach()
if(EA_SANITIZERS)
    if(MSVC)
        message(FATAL_ERROR "EA_SANITIZERS requires GCC or Clang")
    endif()
    foreach(target IN LISTS ea_targets ITEMS ea_sqlite ea_xml)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
endif()
install(TARGETS eventAnalyzer RUNTIME DESTINATION bin)
